\subsection*{Polynomial Regression}

\todo{Explain in your own words why feature scaling, a polynomial basis, and least squares are used here. Interpret the equations below.}

\textbf{Feature standardization.}
Only the training split is used to calculate the mean and standard deviation:
\begin{align}
  \mu_x &= \frac{1}{N_{\mathrm{tr}}}
            \sum_{i=1}^{N_{\mathrm{tr}}}x_i,
  &
  \sigma_x &= \sqrt{\frac{1}{N_{\mathrm{tr}}}
            \sum_{i=1}^{N_{\mathrm{tr}}}(x_i-\mu_x)^2}, \\
  z_i &= \frac{x_i-\mu_x}{\sigma_x},
  &
  N_{\mathrm{tr}} &= 6{,}000.
\end{align}
The same $\mu_x$ and $\sigma_x$ are used to standardize the validation and
test inputs.

\textbf{Polynomial basis.}
For degree $d$, each standardized input is mapped to
\begin{equation}
  \boldsymbol{\phi}_d(z_i)
  = \begin{bmatrix}1 & z_i & z_i^2 & \cdots & z_i^d\end{bmatrix}^{T}.
\end{equation}
The design matrix is
\begin{equation}
  X_d =
  \begin{bmatrix}
    1 & z_1 & z_1^2 & \cdots & z_1^d \\
    1 & z_2 & z_2^2 & \cdots & z_2^d \\
    \vdots & \vdots & \vdots & \ddots & \vdots \\
    1 & z_N & z_N^2 & \cdots & z_N^d
  \end{bmatrix}.
\end{equation}

\textbf{Model and prediction.}
\begin{equation}
  \hat{y}_i
  = \boldsymbol{\phi}_d(z_i)^T\mathbf{w}
  = \sum_{j=0}^{d}w_jz_i^j.
\end{equation}

\textbf{Least-squares solution.}
The weights minimize the mean squared training residual:
\begin{equation}
  J(\mathbf{w})
  = \frac{1}{N_{\mathrm{tr}}}
    \lVert\mathbf{y}-X_d\mathbf{w}\rVert_2^2.
\end{equation}
Setting the gradient to zero gives the normal equation and its solution:
\begin{align}
  X_d^T X_d\hat{\mathbf{w}} &= X_d^T\mathbf{y}, \\
  \hat{\mathbf{w}} &= (X_d^T X_d)^{-1}X_d^T\mathbf{y}.
\end{align}

\textbf{Mean squared error.}
For any split $\mathcal{S}$,
\begin{equation}
  \operatorname{MSE}_{\mathcal{S}}(d)
  = \frac{1}{|\mathcal{S}|}
    \sum_{i\in\mathcal{S}}(y_i-\hat{y}_i)^2.
\end{equation}

\subsection*{Final Model Evaluation}

After selecting $d^*=10$, the training and validation samples are combined:
\begin{equation}
  \mathcal{D}_{\mathrm{fit}}
  = \mathcal{D}_{\mathrm{train}}\cup\mathcal{D}_{\mathrm{val}},
  \qquad |\mathcal{D}_{\mathrm{fit}}|=8{,}000.
\end{equation}
The standardization parameters and degree-10 weights are recalculated using
only $\mathcal{D}_{\mathrm{fit}}$. The resulting held-out test error is
\begin{equation}
  \operatorname{MSE}_{\mathrm{test}}(10)=340.0071.
\end{equation}
